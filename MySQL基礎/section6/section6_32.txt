Q3 
SELECT
TIMESTAMPDIFF(MONTH, '2020-01-01', '2023-03-01') AS months_difference;
TIMESTAMPDIFF は、2つの日付または日時の差を指定した単位で計算し、整数値として返す MySQL の関数です。
単位は年、月、日、時間、分、秒などから選択できます。
月単位で差を計算する場合:

DESC = 降順
文字はシングルクォーテーション''で囲う

SELECT  a.EMPLOYEE_ID , a.EMPLOYEE_NM , a.BIRTH_DATE
  FROM EMPLOYEE_MST a INNER JOIN EMPLOYEE_ENROLLMENT b
  ON a.EMPLOYEE_ID = b.EMPLOYEE_ID
 WHERE b.RETIRED_FLG = 0
   AND TIMESTAMPDIFF(YEAR, BIRTH_DATE, '2017-04-01') <= 40
 ORDER BY a.BIRTH_DATE DESC , a.EMPLOYEE_ID ;

SELECT a.EMPLOYEE_ID  AS EMPLOYEE_ID , a.EMPLOYEE_NM  AS EMPLOYEE_NM ,
  CONCAT (TIMESTAMPDIFF(MONTH , b.ENTERING_DATE , '2017-4-1') DIV 12 ,'年',
    TIMESTAMPDIFF(MONTH , b.ENTERING_DATE , '2017-4-1') MOD 12 ,'ヶ月' ) AS ENROLLMENT_PERIOD_YM
  FROM EMPLOYEE_MST a INNER JOIN EMPLOYEE_ENROLLMENT b
  ON a.EMPLOYEE_ID = b.EMPLOYEE_ID
 WHERE b.RETIRED_FLG = 0 
 ORDER BY TIMESTAMPDIFF(MONTH , b.ENTERING_DATE , '2017-04-01') , a.EMPLOYEE_ID;

SELECT a.EMPLOYEE_ID , a.EMPLOYEE_NM , d.DIVISION_NM
  FROM EMPLOYEE_MST a INNER JOIN EMPLOYEE_ENROLLMENT b
    ON a.EMPLOYEE_ID = b.EMPLOYEE_ID
  INNER JOIN EMPLOYEE_CAREER_HST c
    ON b.EMPLOYEE_ID = c.EMPLOYEE_ID
  INNER JOIN DIVISION_MST        d
    ON c.DIVISION_CD = d.DIVISION_CD
 WHERE b.RETIRED_FLG = 0
   AND '2017-04-01' BETWEEN c.START_DATE AND c.FINISH_DATE
 ORDER BY a.EMPLOYEE_ID ;

Q8
SELECT a.EMPLOYEE_ID  AS EMPLOYEE_ID ,a.EMPLOYEE_NM  AS EMPLOYEE_NM ,
    CASE WHEN c.MANAGER_ID IS NOT NULL THEN 1 ELSE 0 END  AS MANAGER_FLAG
  FROM EMPLOYEE_MST a INNER JOIN EMPLOYEE_ENROLLMENT b
    ON a.EMPLOYEE_ID = b.EMPLOYEE_ID
  LEFT OUTER JOIN DIVISION_MST   c
    ON a.EMPLOYEE_ID = c.MANAGER_ID
 WHERE b.RETIRED_FLG = 0
 ORDER BY a.EMPLOYEE_ID  ;

SELECT COUNT(a.EMPLOYEE_ID) AS COUNT_EMPLOYEE,

    MAX(a.BIRTH_DATE)    AS BIRTH_DATE_YNGEST,
    ROUND( AVG( TIMESTAMPDIFF(YEAR , a.BIRTH_DATE , '2017-04-01') ) , 1)  AS AVERAGE_AGE
  FROM EMPLOYEE_MST a INNER JOIN EMPLOYEE_ENROLLMENT b
    ON a.EMPLOYEE_ID = b.EMPLOYEE_ID
 WHERE b.RETIRED_FLG = 0 ;

SELECT c.DIVISION_CD         AS DIVISION_CD ,
       MAX(c.DIVISION_NM)    AS DIVISION_NM ,
       COUNT(a.EMPLOYEE_ID)  AS COUNT_MEMBER
  FROM EMPLOYEE_ENROLLMENT a INNER JOIN EMPLOYEE_CAREER_HST b
    ON a.EMPLOYEE_ID = b.EMPLOYEE_ID
  INNER JOIN DIVISION_MST        c
    ON b.DIVISION_CD = c.DIVISION_CD
 WHERE a.RETIRED_FLG = 0 
   AND ('2017-04-01' BETWEEN b.START_DATE AND b.FINISH_DATE )
 GROUP BY c.DIVISION_CD
 ORDER BY c.DIVISION_CD ;

Q12
SELECT d.DIVISION_CD       AS DIVISION_CD ,
    MAX(d.DIVISION_NM)  AS DIVISION_NM ,
    ROUND( AVG(TIMESTAMPDIFF(YEAR , a.BIRTH_DATE , '2017-04-01') ) , 1)  AS AVERAGE_AGE
  FROM EMPLOYEE_MST a INNER JOIN EMPLOYEE_ENROLLMENT b
          ON a.EMPLOYEE_ID = b.EMPLOYEE_ID
  INNER JOIN EMPLOYEE_CAREER_HST c
          ON b.EMPLOYEE_ID = c.EMPLOYEE_ID
  INNER JOIN DIVISION_MST        d
          ON c.DIVISION_CD = d.DIVISION_CD
 WHERE b.RETIRED_FLG = 0 
   AND ('2017-04-01' BETWEEN c.START_DATE AND c.FINISH_DATE )
 GROUP BY d.DIVISION_CD
HAVING ROUND( AVG(TIMESTAMPDIFF(YEAR , a.BIRTH_DATE , '2017-04-01') ) , 1) < 37
 ORDER BY d.DIVISION_CD ;

Q13
SELECT a.EMPLOYEE_ID       AS EMPLOYEE_ID ,
    MAX(a.EMPLOYEE_NM)  AS EMPLOYEE_NM ,
    c.DIVISION_CD       AS DIVISION_CD ,
    MAX(c.DIVISION_NM)  AS DIVISION_NM ,
    SUM( CASE WHEN ( '2017-04-01' BETWEEN b.START_DATE AND b.FINISH_DATE )
     THEN TIMESTAMPDIFF(MONTH , b.START_DATE , '2017-04-01')
     ELSE TIMESTAMPDIFF(MONTH , b.START_DATE , b.FINISH_DATE) END  ) AS ENROLLMENT_PERIOD_M
  FROM EMPLOYEE_MST a INNER JOIN EMPLOYEE_CAREER_HST b
          ON a.EMPLOYEE_ID = b.EMPLOYEE_ID
  INNER JOIN DIVISION_MST        c
          ON b.DIVISION_CD = c.DIVISION_CD
 GROUP BY a.EMPLOYEE_ID ,c.DIVISION_CD
 ORDER BY a.EMPLOYEE_ID ,c.DIVISION_CD ;
