SELECT 商品名, 単価, 最終⼊荷⽇
FROM 商品
WHERE単価<= 130

項⽬名に別名を指定する
SELECT 単価AS価格, 商品名FROM 商品
WHERE単価<= 130

項⽬名に計算式を指定する
SELECT 単価* 0.7AS割引価格, 最終⼊荷⽇
FROM 商品
WHERE
商品名= ʼボールペンʼOR 商品名= ʼ鉛筆ʼ
OR 商品名= ʼ消しゴムʼ

項⽬名に定数を指定する
SELECT ʼ130円ʼ AS 価格 FROM 商品
WHERE単価= 130

項⽬名を連結する
SELECT 単価|| ʼ円ʼ AS⾦額FROM 商品

同じ単価が複数⾏あっても１⾏だけ表⽰する//重複を除外
SELECT DISTINCT 単価 FROM 商品

型番がNULLの場合には「なし」にする
SELECT COALESCE(型番, ʼなしʼ) 型番
FROM 商品

それ以外なら「通常価格」と表⽰する

FROM 商品
